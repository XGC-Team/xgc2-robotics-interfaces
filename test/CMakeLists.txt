cmake_minimum_required(VERSION 3.16)
project(interfaces_installed_consumer LANGUAGES C CXX)
find_package(XgcRoboticsInterfaces ${INTERFACES_VERSION} CONFIG REQUIRED PATHS "${INTERFACES_PREFIX}" NO_DEFAULT_PATH)
add_executable(c_smoke c_smoke.c)
set_target_properties(c_smoke PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_link_libraries(c_smoke PRIVATE XgcRoboticsInterfaces::Interfaces)
add_executable(cxx_smoke cxx_smoke.cpp)
set_target_properties(cxx_smoke PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_link_libraries(cxx_smoke PRIVATE XgcRoboticsInterfaces::Interfaces)
